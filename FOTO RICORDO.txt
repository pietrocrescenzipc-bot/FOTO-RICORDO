  <script>
    const photoInput = document.getElementById('photo');
    const previewImg = document.getElementById('preview-img');
    const fileLabel = document.getElementById('file-label');
    const form = document.getElementById('photoForm');
    const modal = document.getElementById('modal');
    const progress = document.getElementById('progress');
    const modalTitle = document.getElementById('modalTitle');
    const modalDesc = document.getElementById('modalDesc');

    // Anteprima Immagine
    photoInput.addEventListener('change', function(e) {
      const file = e.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = function(evt) {
          previewImg.src = evt.target.result;
          previewImg.style.display = 'block';
          fileLabel.innerText = "Foto selezionata: " + file.name;
        }
        reader.readAsDataURL(file);
      }
    });

    // Gestione Invio Form
    form.addEventListener('submit', async function(e) {
      e.preventDefault();

      const name = document.getElementById('name').value;
      const address = document.getElementById('address').value;
      const occasion = document.getElementById('occasion').value;
      const photoFile = photoInput.files[0];

      // Mostra Pop-up
      modal.classList.add('active');
      progress.style.width = '30%';

      let photoUrl = "";

      // Caricamento Immagine su Imgur API
      try {
        const formData = new FormData();
        formData.append('image', photoFile);

        const response = await fetch('https://api.imgur.com/3/image', {
          method: 'POST',
          headers: {
            'Authorization': 'Client-ID 54472a507d3f810' // Client-ID demo anonimo gratuito
          },
          body: formData
        });

        const data = await response.json();
        if (data.success) {
          photoUrl = data.data.link;
        } else {
          photoUrl = "[Foto allegata manualmente]";
        }
      } catch (err) {
        photoUrl = "[Foto allegata manualmente]";
      }

      // Aggiorna Pop-up
      progress.style.width = '100%';
      modalTitle.innerText = "Apertura WhatsApp...";
      modalDesc.innerText = "Premi INVIA nella chat che si aprirà per confermare la stampa a casa!";

      // Costruzione messaggio WhatsApp per 3391318613
      const textMessage = `Ciao The Galleon! 🍕%0AHo compilato la richiesta per la Foto Ricordo stampata a casa:%0A%0A👤 *Nome:* ${encodeURIComponent(name)}%0A🏠 *Indirizzo:* ${encodeURIComponent(address)}%0A🎉 *Occasione:* ${encodeURIComponent(occasion)}%0A📸 *Link Foto:* ${encodeURIComponent(photoUrl)}`;
      
      const whatsappUrl = `https://wa.me/393391318613?text=${textMessage}`;

      // Reindirizzamento WhatsApp dopo 1.2 secondi
      setTimeout(() => {
        window.location.href = whatsappUrl;
      }, 1200);
    });
